GLB модели

Загрузите локальный .glb файл как автобус или карту, либо экспортируйте текущие. Карта автоматически масштабируется под ширину дороги и центрируется. Все материалы переводятся в PBR (MeshStandardMaterial) с применением envMap/IBL.

Физика и разрушение

Пример мода (нажмите, чтобы развернуть) 
// my-mission.js
// Теперь доступно ВСЁ: GAME, THREE, Wheel, Input, EngineSound,
// Minimap, Mods, scene, renderer, camera, window.* — полный контроль.

registerMod({
  name: "Курьер",
  type: "map",
  build({ THREE, GAME, roadCenterX, addMapObject,
          setQuest, setPath, addQuestMarker,
          setMapBounds, scene, renderer }) {

    // 1) Меняем границы игровой карты
    setMapBounds({ minZ: -300, maxZ: 800 });

    // 2) Спавним цель
    const goal = new THREE.Mesh(
      new THREE.BoxGeometry(2, 2, 2),
      new THREE.MeshStandardMaterial({ color: 0xffcb05, emissive: 0x552200 })
    );
    goal.position.set(roadCenterX(200) + 6, 1, 200);
    addMapObject(goal);
    setQuest('Доезжай до жёлтого ящика на Z=200');
    addQuestMarker(goal.position.x, goal.position.z, 'Цель');

    // 3) Рисуем путь
    const path = [];
    for (let z = -50; z <= 200; z += 20)
      path.push({ x: roadCenterX(z), z });
    setPath(path);

    // 4) Полный доступ: правим туман, exposure, масштаб автобуса
    scene.fog.density = 0.0025;
    renderer.toneMappingExposure = 1.25;
    GAME.bus.scale.setScalar(1.05);

    // 5) Меняем миникарту и добавляем свой слой поверх
    Minimap.range = 300;
    (GAME._minimapHooks ||= []).push((ctx, S) => {
      ctx.fillStyle = '#0f0';
      ctx.beginPath(); ctx.arc(10, 10, 4, 0, 6.28); ctx.fill();
    });
  }
});
